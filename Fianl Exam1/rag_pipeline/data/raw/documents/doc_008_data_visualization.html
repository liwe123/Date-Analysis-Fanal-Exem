Matplotlib基础

Matplotlib是Python中最基础的绑图库，可以创建各种静态、动态和交互式图表。

安装和导入

pip install matplotlib

import matplotlib.pyplot as plt
import numpy as np

折线图

x = np.linspace(0, 10, 100)
y = np.sin(x)

plt.figure(figsize=(10, 6))
plt.plot(x, y, label='sin(x)', color='blue', linewidth=2)
plt.xlabel('x')
plt.ylabel('y')
plt.title('正弦函数')
plt.legend()
plt.grid(True)
plt.show()

柱状图

categories = ['A', 'B', 'C', 'D', 'E']
values = [25, 40, 30, 55, 45]

plt.figure(figsize=(8, 6))
plt.bar(categories, values, color=['red', 'green', 'blue', 'yellow', 'purple'])
plt.xlabel('类别')
plt.ylabel('值')
plt.title('柱状图示例')
plt.show()

散点图

x = np.random.randn(100)
y = np.random.randn(100)
colors = np.random.rand(100)
sizes = 1000 * np.random.rand(100)

plt.figure(figsize=(8, 6))
plt.scatter(x, y, c=colors, s=sizes, alpha=0.5, cmap='viridis')
plt.colorbar()
plt.xlabel('x')
plt.ylabel('y')
plt.title('散点图示例')
plt.show()

饼图

labels = ['Python', 'Java', 'C++', 'JavaScript', '其他']
sizes = [30, 25, 20, 15, 10]
explode = (0.1, 0, 0, 0, 0)  # 突出显示第一块

plt.figure(figsize=(8, 8))
plt.pie(sizes, explode=explode, labels=labels, autopct='%1.1f%%',
        shadow=True, startangle=90)
plt.title('编程语言流行度')
plt.show()

Seaborn高级图表

Seaborn是基于Matplotlib的高级统计图表库，提供更美观的默认样式和更高级的图表类型。

安装和导入

pip install seaborn

import seaborn as sns
import pandas as pd

分布图

# 生成随机数据
data = np.random.randn(1000)

# 直方图
plt.figure(figsize=(10, 6))
sns.histplot(data, kde=True)
plt.title('分布图')
plt.show()

箱线图

# 创建示例数据
np.random.seed(42)
data = pd.DataFrame({
    'A': np.random.randn(100),
    'B': np.random.randn(100) + 2,
    'C': np.random.randn(100) - 1
})

plt.figure(figsize=(10, 6))
sns.boxplot(data=data)
plt.title('箱线图')
plt.show()

热力图

# 创建相关矩阵
corr_matrix = data.corr()

plt.figure(figsize=(8, 6))
sns.heatmap(corr_matrix, annot=True, cmap='coolwarm', center=0)
plt.title('相关性热力图')
plt.show()

成对关系图

# 使用内置数据集
iris = sns.load_dataset('iris')
sns.pairplot(iris, hue='species')
plt.show()

Plotly交互式图表

Plotly是一个交互式图表库，可以创建动态、可缩放、可导出的图表。

安装和导入

pip install plotly

import plotly.express as px
import plotly.graph_objects as go

交互式折线图

df = pd.DataFrame({
    'x': np.linspace(0, 10, 100),
    'y': np.sin(np.linspace(0, 10, 100))
})

fig = px.line(df, x='x', y='y', title='交互式正弦函数')
fig.show()

3D散点图

df = pd.DataFrame({
    'x': np.random.randn(100),
    'y': np.random.randn(100),
    'z': np.random.randn(100)
})

fig = px.scatter_3d(df, x='x', y='y', z='z', title='3D散点图')
fig.show()
