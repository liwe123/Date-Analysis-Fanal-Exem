SQL（Structured Query Language）是用于管理关系型数据库的标准语言。

关系型数据库概念：
关系型数据库将数据存储在表中，表由行和列组成。

表结构：
- 表（Table）：数据的集合
- 行（Row）：一条记录
- 列（Column）：一个字段
- 主键（Primary Key）：唯一标识一行的列
- 外键（Foreign Key）：关联两个表的列

SQL数据类型：
1. 整数：INT, SMALLINT, BIGINT
2. 浮点数：FLOAT, DOUBLE, DECIMAL
3. 字符串：VARCHAR, CHAR, TEXT
4. 日期时间：DATE, TIME, DATETIME, TIMESTAMP
5. 布尔：BOOLEAN
6. 二进制：BLOB

创建数据库和表：
-- 创建数据库
CREATE DATABASE mydb;

-- 使用数据库
USE mydb;

-- 创建表
CREATE TABLE users (
    id INT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(50) NOT NULL UNIQUE,
    email VARCHAR(100) NOT NULL,
    age INT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 创建订单表
CREATE TABLE orders (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT,
    product VARCHAR(100),
    amount DECIMAL(10, 2),
    order_date DATE,
    FOREIGN KEY (user_id) REFERENCES users(id)
);

插入数据：
-- 插入单条数据
INSERT INTO users (username, email, age) 
VALUES ('zhangsan', 'zhangsan@example.com', 25);

-- 插入多条数据
INSERT INTO users (username, email, age) VALUES
('lisi', 'lisi@example.com', 30),
('wangwu', 'wangwu@example.com', 28),
('zhaoliu', 'zhaoliu@example.com', 35);

查询数据：
-- 查询所有列
SELECT * FROM users;

-- 查询特定列
SELECT username, email FROM users;

-- 条件查询
SELECT * FROM users WHERE age > 25;

-- 多条件查询
SELECT * FROM users WHERE age > 25 AND city = '北京';

-- 排序
SELECT * FROM users ORDER BY age DESC;

-- 限制结果数量
SELECT * FROM users LIMIT 10;

-- 跳过结果
SELECT * FROM users LIMIT 10 OFFSET 20;

聚合函数：
-- 计数
SELECT COUNT(*) FROM users;

-- 求和
SELECT SUM(amount) FROM orders;

-- 平均值
SELECT AVG(age) FROM users;

-- 最大值
SELECT MAX(age) FROM users;

-- 最小值
SELECT MIN(age) FROM users;

-- 分组聚合
SELECT city, COUNT(*) as user_count 
FROM users 
GROUP BY city;

-- 分组后过滤
SELECT city, COUNT(*) as user_count 
FROM users 
GROUP BY city 
HAVING user_count > 5;

连接查询：
-- 内连接
SELECT users.username, orders.product, orders.amount
FROM users
INNER JOIN orders ON users.id = orders.user_id;

-- 左连接
SELECT users.username, orders.product
FROM users
LEFT JOIN orders ON users.id = orders.user_id;

-- 右连接
SELECT users.username, orders.product
FROM users
RIGHT JOIN orders ON users.id = orders.user_id;

-- 全连接
SELECT users.username, orders.product
FROM users
FULL OUTER JOIN orders ON users.id = orders.user_id;

子查询：
-- 子查询
SELECT * FROM users 
WHERE id IN (SELECT user_id FROM orders WHERE amount > 1000);

-- 相关子查询
SELECT * FROM users u
WHERE EXISTS (SELECT 1 FROM orders o WHERE o.user_id = u.id);

更新数据：
-- 更新单条记录
UPDATE users SET age = 26 WHERE username = 'zhangsan';

-- 更新多条记录
UPDATE users SET city = '上海' WHERE age > 30;

删除数据：
-- 删除单条记录
DELETE FROM users WHERE username = 'zhangsan';

-- 删除多条记录
DELETE FROM users WHERE age < 18;

-- 删除所有数据
DELETE FROM users;

索引：
-- 创建索引
CREATE INDEX idx_username ON users(username);

-- 创建唯一索引
CREATE UNIQUE INDEX idx_email ON users(email);

-- 创建复合索引
CREATE INDEX idx_name_age ON users(username, age);

-- 删除索引
DROP INDEX idx_username ON users;

视图：
-- 创建视图
CREATE VIEW active_users AS
SELECT * FROM users WHERE status = 'active';

-- 使用视图
SELECT * FROM active_users;

事务：
-- 开始事务
START TRANSACTION;

-- 执行SQL语句
INSERT INTO orders (user_id, product, amount) VALUES (1, 'iPhone', 999.99);
UPDATE users SET balance = balance - 999.99 WHERE id = 1;

-- 提交事务
COMMIT;

-- 回滚事务
ROLLBACK;

Python操作数据库：
import sqlite3

# 连接数据库
conn = sqlite3.connect('mydb.db')
cursor = conn.cursor()

# 创建表
cursor.execute('''
    CREATE TABLE IF NOT EXISTS users (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        username TEXT NOT NULL,
        email TEXT NOT NULL
    )
''')

# 插入数据
cursor.execute('INSERT INTO users (username, email) VALUES (?, ?)', 
               ('zhangsan', 'zhangsan@example.com'))

# 查询数据
cursor.execute('SELECT * FROM users')
rows = cursor.fetchall()
for row in rows:
    print(row)

# 提交更改
conn.commit()

# 关闭连接
conn.close()
